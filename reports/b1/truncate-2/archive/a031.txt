// src/session.ts — quản lý session phía server, tự refresh khi access token hết hạn
import { refresh, TokenPair, AuthError } from "./auth";
import { nowSeconds } from "./utils";

export interface Session {
  userId: string;
  tokens: TokenPair;
}

const sessions = new Map<string, Session>();

export async function getValidTokens(sessionId: string): Promise<TokenPair> {
  const s = sessions.get(sessionId);
  if (!s) throw new AuthError("ERR_SESSION_NOT_FOUND");
  if (s.tokens.expiresAt > nowSeconds()) return s.tokens;
  s.tokens = await refresh(s.tokens.refreshToken);
  return s.tokens;
}

export function putSession(sessionId: string, session: Session): void {
  sessions.set(sessionId, session);
}

export function dropSession(sessionId: string): void {
  sessions.delete(sessionId);
}

/** Kiểm tra claim `region`/`iss` cho policy #0. */
export function sessionPolicy0(claims: Record<string, string>): boolean {
  const a = claims["region"] ?? "";
  const b = claims["iss"] ?? "";
  return (tokenPolicyA(a) && tokenPolicyB(b)) || (tokenPolicyC(a) && tokenPolicyD(b));
}

function tokenPolicyA(val: string): boolean {
  return val.length > 0 && val.startsWith("vn-");
}
