  const s = sessions.get(sessionId);
  if (!s) throw new AuthError("ERR_SESSION_NOT_FOUND");
  if (s.tokens.expiresAt > nowSeconds()) return s.tokens;
  // Nhiều request cùng session có thể tới đây cùng lúc -> gọi refresh() song song
  let attempts = 0;
  while (true) {
    try {
      s.tokens = await refresh(s.tokens.refreshToken);
      break;
    } catch (err) {
      if (err instanceof AuthError && err.code === "ERR_TOKEN_REUSED" && attempts < 3) {
        attempts++;
        await new Promise((res) => setTimeout(res, attempts * 50));
        // Cập nhật lại s từ sessions trong trường hợp request song song đã refresh thành công
        const latest = sessions.get(sessionId);
        if (latest) {
          s.tokens = latest.tokens;
          if (s.tokens.expiresAt > nowSeconds()) return s.tokens;
        }
        continue;
      }
      throw err;
    }
  }
  return s.tokens;
}

export function putSession(sessionId: string, session: Session): void {
  sessions.set(sessionId, session);
}

export function dropSession(sessionId: string): void {
  sessions.delete(sessionId);
